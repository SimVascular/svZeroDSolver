20/**
21 * @brief Base class for chamber sub-components (activation functions,
22 * materials, active stress models) that are configured from scalar/vector
23 * input parameters read from the JSON input file.
24 */
25class ParameterHolder {
26 public:

28 * @brief Properties of the input parameters for this component
29 * [(name, InputParameter), ...]
30 */
31 const std::vector<std::pair<std::string, InputParameter>>
32 input_param_properties;
33
34 /**
35 * @brief Construct a ParameterHolder
36 *
37 * @param props Parameter name/spec pairs for this component, used to
38 * initialize params_/params_vec_ with default values
39 */
40 explicit ParameterHolder(
41 const std::vector<std::pair<std::string, InputParameter>>& props);
42
43 virtual ~ParameterHolder() = default;
44
45 /**
46 * @brief Set a scalar parameter value by name.
47 *
48 * Calling function must validate the parameter name and value.
49 *
50 * @param name Parameter name
51 * @param value Parameter value
52 */
53 void set_param(const std::string& name, double value);
54
55 /**
56 * @brief Set a vector-valued parameter value by name.

63 void set_param_vector(const std::string& name,

66 /**
67 * @brief Called once after all parameters have been read from the input

73 virtual void finalize() {}
74
