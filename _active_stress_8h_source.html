23/**
24 * @brief Base class for chamber active stress models
25 *
26 * An ActiveStress owns the dynamics of a chamber's active stress \f$\tau\f$.
27 * Subclasses may introduce additional internal variables and equations
28 * beyond ChamberSphere's fixed core state, and write directly into the same
29 * global SparseSystem that ChamberSphere writes into, using indices that are
30 * a fixed offset into ChamberSphere's own \ref Block::global_var_ids and
31 * \ref Block::global_eqn_ids arrays (which are passed down unchanged).
32 *
33 * ChamberSphere::setup_dofs() always registers, in this order:
34 * internal variables `radius, velo, stress, active_stress, volume,
35 * passive_stress,
36 * <extra_var_names()...>` and equations `momentum, spherical_stress,
37 * volume_change, acceleration, mass_conservation, pressure_equality,
38 * passive_stress, <this model's equations...>`, so the offsets below are
39 * invariant across ActiveStress subclasses.
40 */
41class ActiveStress : public ParameterHolder {
42 public:
43 /**
44 * @name Fixed layout of ChamberSphere's always-present ("core") state
45 *
46 * Indices into the full \ref Block::global_var_ids / \ref
47 * Block::global_eqn_ids arrays that ChamberSphere passes down unchanged.
48 */
49 ///@{
50 static constexpr int RADIUS_VAR =
51 4; ///< Radius position within ChamberSphere's own global_var_ids array
52 static constexpr int VELO_VAR =
53 5; ///< Radial velocity position within ChamberSphere's own
54 ///< global_var_ids array
55 ///<
56 static constexpr int STRESS_VAR =
57 6; ///< Total wall stress position within ChamberSphere's own
58 ///< global_var_ids array
59 ///<
60 static constexpr int TAU_VAR =
61 7; ///< Active stress \f$\tau\f$ position within ChamberSphere's own
62 ///< global_var_ids array
63 ///<
64 static constexpr int VOLUME_VAR =
65 8; ///< Volume position within ChamberSphere's own global_var_ids array
66 static constexpr int PASSIVE_STRESS_VAR =
67 9; ///< Passive wall stress position within ChamberSphere's own
68 ///< global_var_ids array
69 ///<
70 static constexpr int NUM_CORE_VARS =
71 10; ///< A subclass's own extra
72 ///< variables start at
73 ///< global_var_ids[NUM_CORE_VARS + i]
74 static constexpr int NUM_CORE_EQNS =
75 7; ///< A subclass's own extra
76 ///< equations start at
77 ///< global_eqn_ids[NUM_CORE_EQNS + i]
78 ///@}

187/**
188 * @brief Strain-independent active stress model
189 *
190 * Owns 1 extra equation (the \f$\tau\f$ ODE) and 0 extra variables:
191 * \f[
192 * \dot{\tau} + |a| \tau - \sigma_\text{max} a_+ = 0, \quad a_+ = \max(a, 0),
193 * \quad a = f \alpha_\text{max} + (1 - f) \alpha_\text{min}
194 * \f]
195 * where \f$f \in [0, 1]\f$ is the activation function.
196 *
197 * Parameters: `alpha_max`, `alpha_min`, `sigma_max`
198 */
199class StrainIndependentActiveStress : public ActiveStress {
200 public:
201 StrainIndependentActiveStress()
202 : ActiveStress({{"alpha_max", InputParameter()},
203 {"alpha_min", InputParameter()},
204 {"sigma_max", InputParameter()}}) {}
205
206 int num_extra_equations() const override { return 1; }
207
208 ActivationSignalType required_signal_type() const override {
209 return ActivationSignalType::normalized;
210 }

213 const std::vector<int>& global_eqn_ids,
214 const std::vector<int>& global_var_ids) override;
215
216 void update_time(SparseSystem& system, const std::vector<int>& global_eqn_ids,
217 const std::vector<int>& global_var_ids,
218 double activation_signal) override;
219
220 void update_solution(SparseSystem& system,

225 double radius0, double activation_signal) override;

228 double act_ = 0.0; ///< Rate coefficient

232/**
233 * @brief Strain-dependent active stress model from \cite caruel13
234 *
235 * Owns 8 extra equations and 7 extra variables (\f$e_c, \tau_c, k_c,
236 * \omega, \sigma_c, \sigma_s, e_s\f$, in that order):
237 * \f[
238 * \tau = E_s \frac{e_{1D} - e_c}{(1 + 2 e_c)^2}
239 * \f]
240 * with \f$e_{1D}\f$ the strain in fiber direction (a function of the
241 * chamber's radius), and evolution equations
242 * \f[
243 * \dot{\omega} = \frac{1}{\alpha_r}(m_0 - \omega)
244 * \f]
245 * \f[
246 * \dot{e}_c = \frac{1}{\mu}\left(E_s \frac{(e_{1D} - e_c)(1 + 2 e_{1D})}
247 * {(1 + 2 e_c)^3} - \tau_c\right)
248 * \f]
249 * \f[
250 * \dot{k}_c = -(|u|_+ + \omega |u|_- + \alpha |\dot{e}_c|) k_c + n_0 k_0 |u|_+
251 * \f]
252 * \f[
253 * \dot{\tau}_c = -(|u|_+ + \omega |u|_- + \alpha |\dot{e}_c|) \tau_c + n_0
254 * \sigma_0 |u|_+ + k_c \dot{e}_c
255 * \f]
256 * and the additional outputs (sarcomere stress, series element stress and
257 * strain)
258 * \f[
259 * \sigma_c = \tau_c + \mu \dot{e}_c, \quad
260 * \sigma_s = \tau (1 + 2 e_c), \quad
261 * e_s = \frac{\sigma_s}{E_s}
262 * \f]
263 * where \f$n_0\f$ (activation strain-dependence) and \f$m_0\f$ (relaxation
264 * strain-dependence) are piecewise-linear functions of \f$e_c\f$, given as
265 * break point tables (`n0_e_c`/`n0_values` and `m0_e_c`/`m0_values`) and
266 * evaluated with \ref linear_interpolate, and \f$u(t)\f$ is a reaction-rate
267 * signal supplied by the chamber's ActivationFunction (e.g. \ref
268 * PiecewiseRateActivation).
269 *
270 * Parameters: `E_s`, `mu`, `alpha_r`, `alpha`, `k_0`, `sigma_0`, `n0_e_c`,
271 * `n0_values`, `m0_e_c`, `m0_values`
272 */
273class StrainDependentActiveStress : public ActiveStress {
274 public:
275 StrainDependentActiveStress()
276 : ActiveStress({{"E_s", InputParameter()},
277 {"mu", InputParameter()},
278 {"alpha_r", InputParameter()},
279 {"alpha", InputParameter()},
280 {"k_0", InputParameter()},
281 {"sigma_0", InputParameter()},
282 {"n0_e_c", InputParameter(false, true)},
283 {"n0_values", InputParameter(false, true)},
284 {"m0_e_c", InputParameter(false, true)},
285 {"m0_values", InputParameter(false, true)}}) {}
286
287 int num_extra_vars() const override { return 7; }
288 std::list<std::string> extra_var_names() const override {
289 return {"e_c", "tau_c", "k_c", "omega", "sigma_c", "sigma_s", "e_s"};
290 }
291 int num_extra_equations() const override { return 8; }
292
293 ActivationSignalType required_signal_type() const override {
294 return ActivationSignalType::rate;
295 }
296
297 void update_constant(SparseSystem& system,
298 const std::vector<int>& global_eqn_ids,
299 const std::vector<int>& global_var_ids) override;
300
301 void update_time(SparseSystem& system, const std::vector<int>& global_eqn_ids,
302 const std::vector<int>& global_var_ids,
303 double activation_signal) override;
304
305 void update_solution(SparseSystem& system,
306 const std::vector<int>& global_eqn_ids,

308 const Eigen::Matrix<double, Eigen::Dynamic, 1>& y,
309 const Eigen::Matrix<double, Eigen::Dynamic, 1>& dy,
310 double radius0, double activation_signal) override;
311
312 private:
313 /**
314 * @name Extra variable/equation positions within the full ChamberSphere
315 * global_var_ids/global_eqn_ids arrays
316 *
317 * Reproduce the exact global_var_ids/global_eqn_ids indices that the
318 * ChamberSphere used.
319 */
320 ///@{
321 static constexpr int E_C_VAR = NUM_CORE_VARS + 0; // 10
322 static constexpr int TAU_C_VAR = NUM_CORE_VARS + 1; // 11
323 static constexpr int K_C_VAR = NUM_CORE_VARS + 2; // 12
324 static constexpr int OMEGA_VAR = NUM_CORE_VARS + 3; // 13
325 static constexpr int SIGMA_C_VAR = NUM_CORE_VARS + 4; // 14
326 static constexpr int SIGMA_S_VAR = NUM_CORE_VARS + 5; // 15
327 static constexpr int E_S_VAR = NUM_CORE_VARS + 6; // 16
328 static constexpr int TAU_EQN = NUM_CORE_EQNS + 0; // 7
329 static constexpr int OMEGA_EQN = NUM_CORE_EQNS + 1; // 8
330 static constexpr int E_C_EQN = NUM_CORE_EQNS + 2; // 9
331 static constexpr int K_C_EQN = NUM_CORE_EQNS + 3; // 10
332 static constexpr int TAU_C_EQN = NUM_CORE_EQNS + 4; // 11
333 static constexpr int SIGMA_C_EQN = NUM_CORE_EQNS + 5; // 12
334 static constexpr int SIGMA_S_EQN = NUM_CORE_EQNS + 6; // 13
335 static constexpr int E_S_EQN = NUM_CORE_EQNS + 7; // 14
336 ///@}
337
338 /**
339 * @brief Update the strain-dependent coefficients used across this
340 * model's equations
341 *
342 * @param e_c Current sarcomere element strain
343 */
344 void update_active_stress_values(double e_c);
345
346 double n_0_ = 0.0; // activation strain-dependence
347 double m_0_ = 0.0; // relaxation strain-dependence
348 double dn_0_de_c_ = 0.0; // d(n_0)/d(e_c), local slope of n0_e_c/n0_values
349 double dm_0_de_c_ = 0.0; // d(m_0)/d(e_c), local slope of m0_e_c/m0_values
350 double u_plus_ = 0.0; // positive part of the reaction-rate signal
351 double u_minus_ = 0.0; // negative part of the reaction-rate signal
352};
