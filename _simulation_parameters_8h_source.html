22/**
23 * @brief Simulation parameters
24 *
25 * Each of these can be set from the .json configuration file as part of the
26 * dict which is keyed by "simulation_parameters".
27 *
28 * For example:
29 *
30 * "simulation_parameters": {
31 * "number_of_cardiac_cycles": 30,
32 * "number_of_time_pts_per_cardiac_cycle": 201,
33 * "output_all_cycles": true
34 * }
35 *
36 * each parameter in the documentation may be specified using the attribute name
37 * and a value which takes the type of the attribute.
38 *
39 */
40struct SimulationParameters {
41 // Negative value indicates this has not
42 // been read from config file yet.
43 double sim_time_step_size{0.0}; ///< Simulation time step size
44 double sim_abs_tol{0.0}; ///< Absolute tolerance for simulation
45 double sim_cardiac_period{-1.0}; ///< Cardiac period
46 int sim_num_cycles{0}; ///< Number of cardiac cycles to simulate
47 int sim_pts_per_cycle{0}; ///< Number of time steps per cardiac cycle
48 bool use_cycle_to_cycle_error{
49 false}; ///< If model does not have RCR boundary conditions, simulate
50 ///< model to convergence (based on cycle-to-cycle error of last
51 ///< two cardiac cycles); if it does, update number of cardiac
52 ///< cycles to simulate to be value estimated from equation 21 of
53 ///< Pfaller 2021
54 double sim_cycle_to_cycle_error{0}; ///< Cycle-to-cycle error
55 int sim_num_time_steps{0}; ///< Total number of time steps
56 int sim_nliter{0}; ///< Maximum number of non-linear iterations in time
57 ///< integration
58 double sim_rho_infty{0.0}; ///< Spectral radius of generalized-alpha
59 int output_interval{0}; ///< Interval of writing output
60
61 bool sim_steady_initial{0}; ///< Start from steady solution
62 bool output_variable_based{
63 false}; ///< Output variable based instead of vessel based
64 bool output_mean_only{false}; ///< Output only the mean value
65 bool output_derivative{false}; ///< Output derivatives
66 bool output_all_cycles{false}; ///< Output all cardiac cycles
67
68 bool sim_coupled{
69 false}; ///< Running 0D simulation coupled with external solver
70 double sim_external_step_size{0.0}; ///< Step size of external solver if
71 ///< running coupled
72 bool sim_max_iter_error_to_warning{
73 false}; ///< If true, warn instead of throwing when max nonlinear
74 ///< iterations is reached
75};

77/// @brief Wrapper class for nlohmann:json with error checking
78class JsonWrapper : public nlohmann::json {
79 public:
80 /**
81 * @brief Wrap around JSON configuration with detailed error message in case
82 * key is not found in configuration
83 *
84 * @param json JSON configuration
85 * @param component Name of the JSON sub-list to be extracted
86 * @param name_str Name string of the JSON sub-list to be extracted
87 * @param id Index of JSON sub-list to be extracted
88 */
89 JsonWrapper(const nlohmann::json& json, const std::string& component,
90 const std::string& name_str, const int& id)
91 : nlohmann::json(json[component][id]),
92 component(component),
93 name_str(name_str),
94 block_id(id) {}
95
96 /**
97 * @brief Wrap error check around key retrieval (throws detailed error if key
98 * doesn't exist)
99 *
100 * @param key Key to retrieve from JSON object
101 * @return JSON entry of key
102 */
103 const nlohmann::json& operator[](const char* key) const {
104 if (!this->contains(key)) {
105 if (this->contains(name_str)) {
106 const std::string name = this->at(name_str);
107 throw std::runtime_error("Key " + std::string(key) +
108 " not found in element " + name +
109 " of component " + component);
110 } else {
111 throw std::runtime_error(
112 "Key " + std::string(key) + " not found in element number " +
113 std::to_string(block_id) + " of component " + component);
114 }
115 }
116 return this->at(key);
117 }
118
119 // Inherit functions
120 using nlohmann::json::contains;
121 using nlohmann::json::value;
122 using nlohmann::json::operator[];
123
124 private:
125 std::string component;
126 std::string name_str;
127 int block_id;
128};

226/**
227 * @brief Load the 0D block in the model from a configuration
228 *
229 * Every block is created from a type string, read from a key that depends on
230 * the configuration section the block is in:
231 *
232 * | Section | Type key | Valid values documented in |
233 * |---|---|---|
234 * | `vessels` | `zero_d_element_type` | create_vessels() |
235 * | `junctions` | `junction_type` | create_junctions() |
236 * | `boundary_conditions` | `bc_type` | create_boundary_conditions() |
237 * | `external_solver_coupling_blocks` | `type` | create_external_coupling() |
238 * | `closed_loop_blocks` | `closed_loop_type` | create_closed_loop() |
239 * | `valves` | `type` | create_valves() |
240 * | `chambers` | `type` | create_chambers() |
241 *
242 * Each loader checks its type key before creating the block. An invalid type
243 * fails at load time with an error that names the block, section, and key,
244 * and lists the valid values. The loaders derive those values from the block
245 * factory (Model::block_factory_map) filtered by BlockClass, so a newly
246 * registered block is accepted in the section that matches its class without
247 * changes to the loader code. Type strings are case-sensitive.
248 *
249 * The type lists in the comments of the individual loaders are written out by
250 * hand as a reference and can fall behind the registered blocks; the list in
251 * the error message is authoritative.
252 *
253 * External coupling blocks are created before boundary conditions, so
254 * `bc_type` values are validated first, while building the map of boundary
255 * condition types. Otherwise a misspelled `bc_type` on a boundary condition
256 * that a coupling block connects to would be reported as an unrelated
257 * coupling error.
258 *
259 * @param config The json configuration
260 * @param model The 0D model
261 */
262void load_simulation_model(const nlohmann::json& config, Model& model);

291/**
292 * @brief Handle the creation of external coupling blocks and connections with
293 * other blocks
294 *
295 * An external coupling block is the boundary through which an external solver
296 * (e.g. a 3D solver such as svMultiPhysics) exchanges flow and pressure with
297 * the 0D model. Its `type` accepts the same values as `bc_type` (see
298 * create_boundary_conditions()), but use `FLOW` or `PRESSURE`: those are the
299 * blocks whose time series the coupling interface updates
300 * (update_block_params() in interface.cpp), and the error for an invalid
301 * `type` says so.
302 * Besides `name`, `type`, and `values`, each block has:
303 *
304 * - `connected_block`: the 0D block on the other side of the coupling block,
305 * given as a boundary condition's `bc_name`, a vessel's `vessel_name`, or
306 * `ClosedLoopHeartAndPulmonary`.
307 * - `location`: `inlet` or `outlet`, the side of `connected_block` the
308 * coupling block attaches to. With `inlet`, flow goes from the coupling
309 * block into `connected_block`; with `outlet`, from `connected_block` into
310 * the coupling block. The connection to `ClosedLoopHeartAndPulmonary` is
311 * made by create_closed_loop() instead.
312 *
313 * Only some connected blocks are supported at each location. A connected
314 * block's type is its `bc_type` for a boundary condition, or `BloodVessel`
315 * for any vessel. These lists are hard-coded in the implementation
316 * (`supported_types`), so update both together:
317 *
318 * - `inlet`: `RESISTANCE`, `RCR`, `IMPEDANCE`, `ClosedLoopRCR`, `CORONARY`,
319 * `ClosedLoopCoronaryLeft`, `ClosedLoopCoronaryRight`, `BloodVessel`
320 * - `outlet`: `ClosedLoopRCR`, `ClosedLoopHeartAndPulmonary`, `BloodVessel`
321 *
322 * An invalid `type` or `location`, an unknown `connected_block`, or an
323 * unsupported connected block type fails at load time with an error naming
324 * the coupling block.
325 *
326 * @param model The model the block is associated with
327 * @param connections Vector storing the connections between blocks
328 * @param config The JSON configuration
329 * @param component Name of the component to retrieve from config
330 * @param vessel_id_map Map between vessel names and IDs
331 * @param bc_type_map Map between boundary condition names and their types
332 */
333void create_external_coupling(
334 Model& model,
335 std::vector<std::tuple<std::string, std::string>>& connections,
336 const nlohmann::json& config, const std::string& component,
337 std::map<int, std::string>& vessel_id_map,
338 std::map<std::string, std::string>& bc_type_map);

340/**

343 * The block type is read from `bc_type`. Valid values are the blocks of class
