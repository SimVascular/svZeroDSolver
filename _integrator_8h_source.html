16/**
17 * @brief Convergence history of the non-linear iterations
18 *
19 * Holds one entry per time step: the time at the start of the step, the
20 * number of Newton updates it took, the largest absolute residual left in
21 * the solution of that step and the degree of freedom that holds it.
22 */
23struct ConvergenceHistory {
24 std::vector<double> time; ///< Time at the start of the step
25 std::vector<int> iterations; ///< Newton iterations of the step
26 std::vector<double> residual; ///< Largest absolute residual left
27 std::vector<int> residual_dof; ///< DOF index holding that residual
28
29 /**
30 * @brief Record one time step
31 *
32 * @param step_time Time at the start of the step
33 * @param step_iterations Number of Newton updates performed
34 * @param step_residual Largest absolute residual left
35 * @param step_residual_dof DOF index holding that residual
36 */
37 void push_back(double step_time, int step_iterations, double step_residual,
38 int step_residual_dof) {
39 time.push_back(step_time);
40 iterations.push_back(step_iterations);
41 residual.push_back(step_residual);
42 residual_dof.push_back(step_residual_dof);
43 }
44
45 /**
46 * @brief Append another history at the end of this one
47 *
48 * @param other History to append
49 */
50 void append(const ConvergenceHistory& other) {
51 time.insert(time.end(), other.time.begin(), other.time.end());
52 iterations.insert(iterations.end(), other.iterations.begin(),
53 other.iterations.end());
54 residual.insert(residual.end(), other.residual.begin(),
55 other.residual.end());
56 residual_dof.insert(residual_dof.end(), other.residual_dof.begin(),
57 other.residual_dof.end());
58 }
59
60 /**
61 * @brief Discard all recorded time steps
62 */
63 void clear() {
64 time.clear();
65 iterations.clear();
66 residual.clear();
67 residual_dof.clear();
68 }
69
70 /**
71 * @brief Number of recorded time steps
72 *
73 * @return Number of recorded time steps
74 */
75 size_t size() const { return time.size(); }
76};
